    96→    96→function getClientIp(req: Request, server: any): string {
    97→    97→  const xff = req.headers.get('x-forwarded-for')
    98→    98→  if (xff) return xff.split(',')[0].trim()
    99→    99→  const xri = req.headers.get('x-real-ip')
   100→   100→  if (xri) return xri.trim()
   101→   101→  try {
   102→   102→    const ip = server.requestIP(req) as { address: string } | null
   103→   103→    if (ip?.address) return ip.address
   104→   104→  } catch {
   105→   105→    /* ignore */
   106→   106→  }
   107→   107→  return 'unknown'
   108→   108→}

   110→   110→function verifyHmac(rawBody: string, signature: string, secret: string): boolean {
   111→   111→  const expected = createHmac('sha256', secret).update(rawBody).digest('hex')
   112→   112→  const a = Buffer.from(signature)
   113→   113→  const b = Buffer.from(expected)
   114→   114→  if (a.length !== b.length) return false
   115→   115→  return timingSafeEqual(a, b)
   116→   116→}

   144→   144→  const digits = String(val).replace(/[^\d]/g, '')

   751→   751→async function handleWebhook(req: Request, server: any): Promise<Response> {
   752→   752→  // ─── 1. IP allowlist check ──────────────────────────────────────────────
   753→   753→  const clientIp = getClientIp(req, server)
   754→   754→  if (VAPI_IP_ALLOWLIST_RAW.trim().length > 0) {
   755→   755→    const allowlist = VAPI_IP_ALLOWLIST_RAW.split(',').map((s) => s.trim()).filter(Boolean)
   756→   756→    const allowed = allowlist.includes('*') ||
   757→   757→      allowlist.some((a) => clientIp === a || clientIp.endsWith(`.${a}`))
   758→   758→    if (!allowed) {
   759→   759→      console.warn(`[vapi-webhook] 403 IP not in allowlist: ${clientIp}`)
   760→   760→      return json({ ok: false, error: 'IP not allowed', ip: clientIp }, 403)
   761→   761→    }
   762→   762→  } else {
   763→   763→    console.warn('[vapi-webhook] WARNING: VAPI_IP_ALLOWLIST not set — allowing all IPs (dev only).')
   764→   764→  }
   765→   765→
   766→   766→  // ─── 2. Read raw body (preserve for HMAC) ──────────────────────────────
   767→   767→  const rawBody = await req.text()
   768→   768→
   769→   769→  // ─── 3. HMAC-SHA256 verify ──────────────────────────────────────────────
   770→   770→  if (DEV_MODE) {
   771→   771→    const reason = VAPI_SECRET === '' ? 'VAPI_SECRET unset' : 'VAPI_SECRET=dev-secret'
   772→   772→    console.warn(`[vapi-webhook] WARNING: DEV MODE — HMAC verification SKIPPED (${reason}).`)
   773→   773→  } else {
   774→   774→    const sig = req.headers.get('x-vapi-signature') || ''
   775→   775→    if (!sig) return json({ ok: false, error: 'Missing signature' }, 401)
   776→   776→    if (!verifyHmac(rawBody, sig, VAPI_SECRET)) {
   777→   777→      return json({ ok: false, error: 'Invalid signature' }, 401)
   778→   778→    }
   779→   779→  }
   780→   780→
   781→   781→  // ─── 4. Parse JSON payload ──────────────────────────────────────────────
   782→   782→  let payload: any
   783→   783→  try {
   784→   784→    payload = JSON.parse(rawBody)
   785→   785→  } catch {
   786→   786→    return json({ ok: false, error: 'Invalid JSON body' }, 400)
   787→   787→  }
   788→   788→
   789→   789→  const msgType = str(payload?.message?.type)
   790→   790→  if (msgType && msgType !== 'end-of-call-report') {
   791→   791→    return json({ ok: true, ignored: msgType })
   792→   792→  }
   793→   793→
   794→   794→  const message = payload?.message ?? {}
   795→   795→  const call = message?.call
   796→   796→  if (!call || typeof call !== 'object' || !call.id) {
   797→   797→    return json({ ok: false, error: 'Missing message.call.id' }, 400)
   798→   798→  }
   799→   799→
   800→   800→  const vapiCallId = str(call.id)
   801→   801→  const callerPhone = str(call?.customer?.number)
   802→   802→  const transcript = str(call?.transcript) || str(message?.transcript) || str(message?.artifact?.transcript)
   803→   803→  const structured = ((call?.analysis?.structuredData || message?.analysis?.structuredData || {}) as Record<string, unknown>)
   804→   804→
   805→   805→  function pick(...keys: string[]): string {
   806→   806→    for (const k of keys) {
   807→   807→      const v = structured[k]
   808→   808→      if (v) return str(v)
   809→   809→    }
   810→   810→    return ''
   811→   811→  }
   812→   812→
   813→   813→  const callerName = pick('caller_name', 'callerName', 'name') || 'अनाम'
   814→   814→  const callReason = pick('call_reason', 'complaintReason', 'callReason', 'description') || transcript
   815→   815→  const categoryRich = pick('category', 'categoryCode')
   816→   816→  const categoryShort = toShortCategory(categoryRich)
   817→   817→  const priority = pick('priority') || 'medium'
   818→   818→  const departmentCode = pick('departmentCode', 'department_code') || 'general'
   819→   819→  const wardNumber = num(pick('wardNumber', 'ward_number', 'ward'))
   820→   820→  const villageName = pick('villageName', 'village_name', 'village')
   821→   821→  const transferStatus = pick('transferStatus', 'transfer_status') || 'pending'
   822→   822→  const officerName = pick('officerName', 'officer_name')
   823→   823→  const officerPhone = pick('officerPhone', 'officer_phone')
   824→   824→
   825→   825→  if (!callReason) {
   826→   826→    return json({ ok: false, error: 'Missing call_reason / transcript' }, 400)
   827→   827→  }
   828→   828→
   829→   829→  // ─── 5. Idempotency check on vapiCallId ────────────────────────────────
   830→   830→  const existing = await db.complaint.findFirst({
   831→   831→    where: { vapiCallId },
   832→   832→    select: { id: true, trackingId: true },
   833→   833→  }).catch((e) => {
   834→   834→    console.error('[vapi-webhook] DB error during dedup check:', e)
   835→   835→    return null
   836→   836→  })
   837→   837→  if (existing) {
   838→   838→    console.log(`[vapi-webhook] DEDUPED — complaint already exists for vapiCallId=${vapiCallId}`)
   839→   839→    return json({ ok: true, deduped: true, trackingId: existing.trackingId })
   840→   840→  }
   841→   841→
   842→   842→  // ─── 6. Generate tracking ID + insert ──────────────────────────────────
   843→   843→  const trackingId = generateTrackingId()
   844→   844→  const nowIso = new Date().toISOString()
   845→   845→  const timeline = [
   846→   846→    {
   847→   847→      status: 'Pending',
   848→   848→      ts: nowIso,
   849→   849→      note: `AI voice assistant filed complaint (priority=${priority}, dept=${departmentCode})`,
   850→   850→      by: 'vapi-assistant',
   851→   851→    },
   852→   852→  ]
   853→   853→

   855→   855→  const routingRule = await db.routingRule.findUnique({ where: { category: categoryRich || categoryShort } }).catch(() => null)
   856→   856→
   857→   857→  const complaint = await db.complaint.create({
   858→   858→    data: {
   859→   859→      trackingId,
   860→   860→      vapiCallId,
   861→   861→      callerName,
   862→   862→      callerPhone,
   863→   863→      callReason,
   864→   864→      category: categoryShort,
   865→   865→      status: 'Pending',
   866→   866→      rawTranscript: transcript,
   867→   867→      timeline: JSON.stringify(timeline),
   868→   868→    },
   869→   869→  })
   870→   870→
   871→   871→  // Insert the CallRecord (rich data)
   872→   872→  const callRecord = await db.callRecord.create({
   873→   873→    data: {
   874→   874→      vapiCallId,
   875→   875→      callerPhone: callerPhone || null,
   876→   876→      citizenName: callerName,
   877→   877→      villageName: villageName || null,
   878→   878→      wardNumber: wardNumber,
   879→   879→      category: categoryRich || categoryShort,
   880→   880→      priority,
   881→   881→      departmentCode,
   882→   882→      transferTarget: officerPhone || null,
   883→   883→      transferStatus,
   884→   884→      complaintId: complaint.id,
